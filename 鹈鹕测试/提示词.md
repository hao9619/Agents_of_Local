    let start = null;
    
    function animate(timestamp) {
        if (start === null) {
            start = timestamp;
        }
        
        const elapsed = (timestamp - start) / 1000;
        const theta = elapsed * 2 * Math.PI / 1.6; // 1.6 seconds per rotation

        // --- Spokes Animation ---
        const rearSpokes = document.getElementById('rearSpokes');
        const frontSpokes = document.getElementById('frontSpokes');
        
        // Rotate spokes in degrees (SVG rotate attribute)
        const deg = theta * 180 / Math.PI;
        rearSpokes.setAttribute('transform', `rotate(${deg})`);
        frontSpokes.setAttribute('transform', `rotate(${deg})`);

        // --- Pedals & Legs Animation ---
        const crankHubX = 440;
        const crankHubY = 420;
        const pedalRadius = 28;
        
        // Near Leg (Right side)
        const nearX = crankHubX + pedalRadius * Math.cos(theta);
        const nearY = crankHubY + pedalRadius * Math.sin(theta);
        
        // Far Leg (Left side) - theta + PI
        const farTheta = theta + Math.PI;
        const farX = crankHubX + pedalRadius * Math.cos(farTheta);
        const farY = crankHubY + pedalRadius * Math.sin(farTheta);

        // Update Pedals
        const nearPedal = document.getElementById('nearPedal');
        const farPedal = document.getElementById('farPedal');
        nearPedal.setAttribute('cx', nearX);
        nearPedal.setAttribute('cy', nearY);
        farPedal.setAttribute('cx', farX);
        farPedal.setAttribute('cy', farY);

        // Update Crank Arm (Line between pedals)
        const crankArm = document.getElementById('crankArm');
        crankArm.setAttribute('x1', nearX);
        crankArm.setAttribute('y1', nearY);
        crankArm.setAttribute('x2', farX);
        crankArm.setAttribute('y2', farY);

        // Update Legs (Polyline: Hip -> Knee -> Foot)
        const hipX = 420;
        const hipY = 315;
        
        // Near Leg Points: Hip(420,315), Knee(midpoint+shift), Foot(nearX,nearY)
        const nearKneeX = (hipX + nearX) / 2 + 35; 
        const nearKneeY = (hipY + nearY) / 2;
        const nearLegPoints = `${hipX},${hipY} ${nearKneeX},${nearKneeY} ${nearX},${nearY}`;
        
        // Far Leg Points: Hip(420,315), Knee(midpoint+shift), Foot(farX,farY)
        const farKneeX = (hipX + farX) / 2 - 35; 
        const farKneeY = (hipY + farY) / 2;
        const farLegPoints = `${hipX},${hipY} ${farKneeX},${farKneeY} ${farX},${farY}`;

        document.getElementById('nearLeg').setAttribute('points', nearLegPoints);
        document.getElementById('farLeg').setAttribute('points', farLegPoints);

        // --- Road Marks Animation ---
        const roadMarks = document.getElementById('roadMarks');
        // Smoothly looping negative x translation using elapsed
        const offset = (elapsed % 20) * -1; 
        roadMarks.setAttribute('transform', `translate(${offset}, 0)`);

        requestAnimationFrame(animate);
    }
